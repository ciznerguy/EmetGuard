כל פעולה מחזירה מחרוזת: הודעת שגיאה, או מחרוזת ריקה אם הכול תקין

public static string ValidateUsername(string? username)
{
    if (string.IsNullOrWhiteSpace(username))
    {
        return "יש להקליד שם משתמש.";
    }
    string value = username.Trim();
    if (value.Length < UsernameMinLength || value.Length > UsernameMaxLength)
    {
        return $"...בין {UsernameMinLength} ל-{UsernameMaxLength} תווים.";
    }
    foreach (char c in value)
    {
        bool isEnglishLetter = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z');
        bool isDigit = c >= '0' && c <= '9';
        if (!isEnglishLetter && !isDigit && c != '_')
        {
            return "...רק אותיות באנגלית, ספרות וקו תחתון.";
        }
    }
    return "";
}

הקוד: בדיקת משתמש שלם

הפעולה Validate אוספת את כל השגיאות לרשימה

public static List<string> Validate(User user)
{
    List<string> errors = new List<string>();

    string error = ValidateUsername(user.Username);
    if (error != "") { errors.Add(error); }

    error = ValidateEmail(user.Email);
    if (error != "") { errors.Add(error); }

    error = ValidatePassword(user.Password);
    if (error != "") { errors.Add(error); }

    return errors;
}
